binary representation
    bit single binary digit
    packaged in 8
        8 bits is a byte
    base 2
    each place increases by 8
    little endian
        right to left
    big endian
        left to right
    << n shifts right n times
    >> shifts left n times
        if flows past 8 either left or right, truncate
    xor - exclusive or
        only one can be true
    or 
        both or one can be true
    useful for storing multiple flags in one byte
    file permision
        3 num - dictate who can do what
            first num - owner capabilites
            second num - group capabilities
            third num - everyone else capabilites

class syntax
{
    class Car {
    private:
    std::string brand;
    int year;

    public:
    // 1. Default Constructor (called when no arguments are provided)
    Car() {
        brand = "Unknown";
        year = 0;
        std::cout << "Default constructor called!" << std::endl;
    }

    // 2. Parameterized Constructor (called when arguments are provided)
    // Using a member initializer list ': brand(x), year(y)' is the preferred C++ way
    Car(std::string carBrand, int carYear) : brand(carBrand), year(carYear) {
        std::cout << "Parameterized constructor called!" << std::endl;
    }

    // Method to display car details
    void displayInfo() {
        std::cout << "Brand: " << brand << ", Year: " << year << "\n" << std::endl;
    }
    };

    int main() {
         // Triggers the Default Constructor
        Car car1; 
        car1.displayInfo();

        // Triggers the Parameterized Constructor
        Car car2("Toyota", 2024); 
        car2.displayInfo();

        return 0;
        }
}
